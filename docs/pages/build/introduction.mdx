import React, { useState, useEffect } from 'react';
import {
  StyleSheet,
  Text,
  View,
  TouchableOpacity,
  SafeAreaView,
  StatusBar,
  Modal,
  Alert,
} from 'react-native';
import LinearGradient from 'react-native-linear-gradient';
import Animated, { FadeInUp, FadeInDown, ZoomIn } from 'react-native-reanimated';
import Purchases from 'react-native-purchases';
import Superwall from '@superwall/react-native-superwall';

import { Question, LifelineType } from '../types/game';

// Sample Level Ladder Values
const LADDER_REWARDS = [100, 200, 300, 500, 1000, 2000, 4000, 8000, 16000, 32000, 64000, 125000, 250000, 500000, 1000000];

const SAMPLE_QUESTION: Question = {
  question_id: 'sp_q_5082',
  category: 'Biblical History & Events',
  difficulty_level: 5,
  story_context: 'During the reign of King Hezekiah, Jerusalem faced an imminent siege by the Assyrian army under Sennacherib.',
  prompt: 'Which strategic engineering project did Hezekiah undertake to secure the city\'s water supply prior to the siege?',
  options: [
    { id: 'A', text: 'Rebuilding the Eastern Water Gate' },
    { id: 'B', text: 'Tunneling through solid rock from Gihon Spring to Siloam' },
    { id: 'C', text: 'Constructing deep reservoirs in the Kidron Valley' },
    { id: 'D', text: 'Diverting the Jordan River into northern cisterns' },
  ],
  correct_option_id: 'B',
  scripture_proof: '2 Chronicles 32:30',
  reasoning_explanation: 'Hezekiah blocked the upper outlet of the Gihon spring and channeled the water straight down to the west side of the City of David.',
  lifeline_data: {
    scriptural_hint: 'Focus on the underground channel leading to Siloam.',
    fifty_fifty_remove: ['A', 'D'],
  },
};

export const QuizArenaScreen: React.FC = () => {
  const [questionIndex, setQuestionIndex] = useState<number>(4); // Question 5 (Index 4) triggers Council Offer
  const [disabledOptions, setDisabledOptions] = useState<string[]>([]);
  const [usedLifelines, setUsedLifelines] = useState<Record<LifelineType, boolean>>({
    scriptural_hint: false,
    fifty_fifty: false,
    verse_swap: false,
    community_wisdom: false,
  });
  const [showCouncilModal, setShowCouncilModal] = useState<boolean>(false);
  const [activeHint, setActiveHint] = useState<string | null>(null);

  // Trigger Council Offer on Milestone Checkpoints (e.g., Question 5 & 10)
  useEffect(() => {
    if (questionIndex === 4 || questionIndex === 9) {
      setShowCouncilModal(true);
    }
  }, [questionIndex]);

  const handleAnswer = (selectedId: string) => {
    if (selectedId === SAMPLE_QUESTION.correct_option_id) {
      if (questionIndex < LADDER_REWARDS.length - 1) {
        setQuestionIndex(prev => prev + 1);
        setDisabledOptions([]);
        setActiveHint(null);
      } else {
        Alert.alert('Victory!', 'You have mastered The Ascent!');
      }
    } else {
      // Trigger Superwall Paywall on Failure / Second Chance
      Superwall.shared.register('game_over_second_chance');
      Alert.alert('Ascent Failed', `Correct answer: ${SAMPLE_QUESTION.correct_option_id}. ${SAMPLE_QUESTION.reasoning_explanation}`);
    }
  };

  const useFiftyFifty = () => {
    if (usedLifelines.fifty_fifty) return;
    setDisabledOptions(SAMPLE_QUESTION.lifeline_data.fifty_fifty_remove);
    setUsedLifelines(prev => ({ ...prev, fifty_fifty: true }));
  };

  const useScripturalHint = () => {
    if (usedLifelines.scriptural_hint) return;
    setActiveHint(SAMPLE_QUESTION.lifeline_data.scriptural_hint);
    setUsedLifelines(prev => ({ ...prev, scriptural_hint: true }));
  };

  const handleAcceptCouncilOffer = () => {
    setShowCouncilModal(false);
    const guaranteedPearls = LADDER_REWARDS[questionIndex - 1] || 500;
    Alert.alert('Cashed Out!', `You walked away with ${guaranteedPearls} Pearls.`);
  };

  return (
    <LinearGradient colors={['#080A15', '#161B3D', '#080A15']} style={styles.container}>
      <StatusBar barStyle="light-content" />
      <SafeAreaView style={styles.safeArea}>
        
        {/* Top Header HUD */}
        <View style={styles.headerContainer}>
          <View style={styles.badgeContainer}>
            <Text style={styles.badgeText}>CHERUBIM LEAGUE</Text>
          </View>
          <View style={styles.rewardContainer}>
            <Text style={styles.rewardLabel}>CURRENT PRIZE</Text>
            <Text style={styles.rewardValue}>{LADDER_REWARDS[questionIndex]} Pearls</Text>
          </View>
        </View>

        {/* Lifelines Bar */}
        <View style={styles.lifelinesContainer}>
          <TouchableOpacity 
            style={[styles.lifelineButton, usedLifelines.fifty_fifty && styles.lifelineDisabled]} 
            onPress={useFiftyFifty}
            disabled={usedLifelines.fifty_fifty}>
            <Text style={styles.lifelineText}>50:50</Text>
          </TouchableOpacity>
          <TouchableOpacity 
            style={[styles.lifelineButton, usedLifelines.scriptural_hint && styles.lifelineDisabled]} 
            onPress={useScripturalHint}
            disabled={usedLifelines.scriptural_hint}>
            <Text style={styles.lifelineText}>Hint</Text>
          </TouchableOpacity>
        </View>

        {/* Question & Story Context Card */}
        <Animated.View entering={FadeInUp.duration(600)} style={styles.glassCard}>
          <Text style={styles.categoryText}>{SAMPLE_QUESTION.category.toUpperCase()}</Text>
          {SAMPLE_QUESTION.story_context && (
            <Text style={styles.storyText}>{SAMPLE_QUESTION.story_context}</Text>
          )}
          <Text style={styles.promptText}>{SAMPLE_QUESTION.prompt}</Text>
          
          {activeHint && (
            <View style={styles.hintContainer}>
              <Text style={styles.hintText}>💡 {activeHint}</Text>
            </View>
          )}
        </Animated.View>

        {/* Options Grid */}
        <View style={styles.optionsContainer}>
          {SAMPLE_QUESTION.options.map((option, idx) => {
            const isDisabled = disabledOptions.includes(option.id);
            return (
              <Animated.View key={option.id} entering={FadeInDown.delay(idx * 100)}>
                <TouchableOpacity
                  activeOpacity={0.8}
                  disabled={isDisabled}
                  style={[styles.optionButton, isDisabled && styles.optionDisabled]}
                  onPress={() => handleAnswer(option.id)}>
                  <Text style={styles.optionPrefix}>{option.id}</Text>
                  <Text style={styles.optionText}>{option.text}</Text>
                </TouchableOpacity>
              </Animated.View>
            );
          })}
        </View>

        {/* The Council's Offer (Banker) Modal */}
        <Modal visible={showCouncilModal} transparent animationType="fade">
          <View style={styles.modalOverlay}>
            <Animated.View entering={ZoomIn.duration(400)} style={styles.councilCard}>
              <Text style={styles.councilTitle}>THE COUNCIL'S OFFER</Text>
              <Text style={styles.councilDescription}>
                You have reached Question {questionIndex + 1}. The Council offers you a guaranteed cash out before you risk your progression.
              </Text>
              <Text style={styles.offerAmount}>{LADDER_REWARDS[questionIndex - 1]} Pearls</Text>
              
              <TouchableOpacity style={styles.acceptButton} onPress={handleAcceptCouncilOffer}>
                <Text style={styles.acceptButtonText}>WALK AWAY WITH PEARLS</Text>
              </TouchableOpacity>
              
              <TouchableOpacity style={styles.declineButton} onPress={() => setShowCouncilModal(false)}>
                <Text style={styles.declineButtonText}>RISK IT ALL & CONTINUE</Text>
              </TouchableOpacity>
            </Animated.View>
          </View>
        </Modal>

      </SafeAreaView>
    </LinearGradient>
  );
};
